<script>
  // Floating leaves (instead of hearts)
  const leavesLayer = document.getElementById("leaves");
  const LEAVES = ["🍃","🌿","✨","🌙","🍂"];

  function spawnLeaf(intense=false){
    const el = document.createElement("div");
    el.className="leaf";
    el.textContent = LEAVES[Math.floor(Math.random()*LEAVES.length)];
    el.style.left = (Math.random()*100) + "vw";

    const drift = (Math.random()*180 - 90).toFixed(0) + "px";
    const scale = intense ? (0.95 + Math.random()*1.3) : (0.55 + Math.random()*0.85);
    el.style.setProperty("--drift", drift);
    el.style.setProperty("--scale", scale.toFixed(2));

    const base = intense ? 22 : 14;
    el.style.fontSize = (base + Math.random()*(intense ? 22 : 10)).toFixed(0) + "px";
    el.style.animationDuration = (intense ? (4+Math.random()*4) : (8+Math.random()*8)).toFixed(2) + "s";

    leavesLayer.appendChild(el);
    el.addEventListener("animationend", ()=> el.remove());
  }

  setInterval(()=>spawnLeaf(false), 520);

  function burst(count=70){
    for(let i=0;i<count;i++){
      setTimeout(()=>spawnLeaf(true), i*25);
    }
  }

  // NO BUTTON TELEPORT
  const noBtn = document.getElementById("noBtn");
  function rand(min,max){ return Math.floor(Math.random()*(max-min+1))+min; }

  function teleportNo(){
    const rect = noBtn.getBoundingClientRect();
    const padding = 12;

    const maxX = window.innerWidth - rect.width - padding;
    const maxY = window.innerHeight - rect.height - padding;

    const x = rand(padding, Math.max(padding, maxX));
    const y = rand(padding, Math.max(padding, maxY));

    noBtn.classList.remove("teleport");
    void noBtn.offsetWidth;

    noBtn.style.left = x + "px";
    noBtn.style.top  = y + "px";
    noBtn.style.transform = "translate(0,0)";

    noBtn.classList.add("teleport");
    burst(12);
  }

  noBtn.addEventListener("click", (e)=>{
    e.preventDefault();
    teleportNo();
  });

  window.addEventListener("resize", teleportNo);

  // YES -> WIPE -> PAGE2
  const yesBtn = document.getElementById("yesBtn");
  const wipe = document.getElementById("wipe");
  const page1 = document.getElementById("page1");
  const page2 = document.getElementById("page2");
  const backBtn = document.getElementById("backBtn");

  yesBtn.addEventListener("click", ()=>{
    burst(120);
    wipe.classList.add("show");

    setTimeout(()=>{
      page1.style.display="none";
      page2.classList.add("show");
    }, 520);

    setTimeout(()=>{
      wipe.classList.remove("show");
    }, 1100);
  });

  backBtn.addEventListener("click", ()=>{
    page2.classList.remove("show");
    page1.style.display="";
    burst(50);
    teleportNo();
  });

  teleportNo();
</script>
